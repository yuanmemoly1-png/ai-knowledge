# 基础语法速查

> [!note] 怎么用这份速查
> 每一节都是可直接运行的最小示例。建议全部亲手敲一遍，改掉变量值再跑一次，观察输出变化。

## 1. 变量与类型

Python 是动态类型：不用声明类型，赋值即创建。

```python
name = "Kimi"        # str 字符串
age = 2              # int 整数
price = 9.9          # float 浮点数
is_active = True     # bool 布尔值
nothing = None       # NoneType，表示"空"

print(type(name))    # <class 'str'>

# 类型转换
num = int("42")          # 字符串 → 整数
text = str(3.14)         # 浮点 → 字符串
score = float("9.5")     # 字符串 → 浮点

# f-string 格式化（AI 开发中拼 Prompt 天天用）
model = "gpt-4o"
tokens = 128
prompt = f"模型 {model} 本次消耗了 {tokens} 个 token"
print(prompt)
```

## 2. 字符串常用方法

```python
s = "  Hello, AI World  "

s.strip()            # 去掉首尾空白 → "Hello, AI World"
s.lower()            # 全小写
s.upper()            # 全大写
"hello".replace("l", "L")       # → "heLLo"
"a,b,c".split(",")              # → ["a", "b", "c"]
"-".join(["a", "b", "c"])       # → "a-b-c"
"hello world".startswith("he")  # → True
"report.pdf".endswith(".pdf")   # → True
"AI" in "I love AI"             # → True
len("hello")                    # → 5

# 切片：字符串[起点:终点:步长]，终点不含
text = "PromptEngineering"
print(text[0:6])    # Prompt
print(text[-11:])   # Engineering（负数从尾部数）
print(text[::-1])   # gnireenignEtpromP（反转）

# 多行字符串（写 Prompt 模板的标准姿势）
template = """你是一个翻译助手。
请把以下内容翻译成{lang}：
{content}"""
print(template.format(lang="英语", content="你好"))
```

## 3. 条件与循环

```python
# if / elif / else
score = 85
if score >= 90:
    level = "优秀"
elif score >= 60:
    level = "及格"
else:
    level = "不及格"
print(level)  # 及格

# for 循环遍历列表
models = ["gpt-4o", "claude-sonnet", "kimi-k2"]
for m in models:
    print(f"测试模型: {m}")

# range 生成数字序列
for i in range(3):      # 0, 1, 2
    print(f"第 {i + 1} 次重试")

# while 循环 + break/continue
attempt = 0
while True:
    attempt += 1
    if attempt >= 3:
        print("重试次数用尽")
        break
    print(f"尝试第 {attempt} 次")
```

## 4. 列表/字典推导式

一行生成新列表/字典，Python 代码里出现频率极高。

```python
nums = [1, 2, 3, 4, 5, 6]

# 列表推导式：[表达式 for 变量 in 序列 if 条件]
squares = [n ** 2 for n in nums]               # [1, 4, 9, 16, 25, 36]
evens = [n for n in nums if n % 2 == 0]        # [2, 4, 6]

# 字典推导式
prices = {"apple": 3, "banana": 2, "cherry": 8}
expensive = {k: v for k, v in prices.items() if v > 2}
print(expensive)  # {'apple': 3, 'cherry': 8}

# 实战：清洗一段模型输出里的空行
lines = ["第一段", "", "  ", "第二段"]
clean = [line.strip() for line in lines if line.strip()]
print(clean)  # ['第一段', '第二段']
```

## 5. 异常处理 try/except

调 API、读文件必用。原则：**只捕获你预期的异常，别裸 `except:`**。

```python
# 基本结构
def parse_int(text):
    try:
        return int(text)
    except ValueError:
        print(f"'{text}' 不是合法数字，返回默认值 0")
        return 0

print(parse_int("42"))    # 42
print(parse_int("abc"))   # 'abc' 不是合法数字，返回默认值 0

# 完整四件套：try / except / else / finally
def divide(a, b):
    try:
        result = a / b
    except ZeroDivisionError:
        print("除数不能为 0")
        return None
    else:
        # 没出异常才执行
        return result
    finally:
        # 无论是否异常都会执行（常用于释放资源）
        print("计算结束")

divide(10, 2)   # 计算结束 → 返回 5.0
divide(10, 0)   # 除数不能为 0 → 计算结束

# 主动抛异常
def call_api(key):
    if not key:
        raise ValueError("API Key 不能为空")
    return "ok"
```

## 6. 文件读写 with open

`with` 语句会在代码块结束后自动关闭文件，永远比手动 `f.close()` 安全。

```python
# 写文件（'w' 覆盖写，'a' 追加写）
with open("notes.txt", "w", encoding="utf-8") as f:
    f.write("第一行：Python 文件读写\n")
    f.write("第二行：记得指定 encoding='utf-8'\n")

# 读整个文件
with open("notes.txt", "r", encoding="utf-8") as f:
    content = f.read()
print(content)

# 按行读（大文件推荐，不占内存）
with open("notes.txt", "r", encoding="utf-8") as f:
    for line in f:
        print("读到:", line.strip())

# 读 JSON 配置（AI 项目里配置文件基本都是 JSON）
import json

with open("config.json", "w", encoding="utf-8") as f:
    json.dump({"model": "gpt-4o", "temperature": 0.7}, f, ensure_ascii=False)

with open("config.json", "r", encoding="utf-8") as f:
    config = json.load(f)
print(config["model"])  # gpt-4o
```

> [!warning] Windows 中文乱码
> Windows 默认编码是 GBK，读写文件不指定 `encoding="utf-8"` 时，中文极易乱码或报 `UnicodeDecodeError`。养成永远写 `encoding="utf-8"` 的习惯。
